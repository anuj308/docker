30. Puppet-manage HAProxy
Add this to site.pp:
  file { '/etc/haproxy/haproxy.cfg':
    ensure  => file,
    content => 'global
    log /dev/log local0
    log /dev/log local1 notice
    daemon
 
defaults
    log global
    mode http
    option httplog
    option dontlognull
    timeout connect 5s
    timeout client  30s
    timeout server  30s
 
frontend web_frontend
    bind *:80
    default_backend web_servers
 
backend web_servers
    balance roundrobin
    option httpchk GET /
    server web1 127.0.0.1:8081 check
    server web2 127.0.0.1:8082 check
',
    notify => Service['haproxy'],
  }
